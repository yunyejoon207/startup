<script type="py">
from js import document, localStorage
import json

ranking = json.loads(localStorage.getItem("ranking") or "[]")

def save():
    localStorage.setItem("ranking", json.dumps(ranking))

def render():
    box = document.getElementById("list")
    ranking.sort(key=lambda x:x["score"], reverse=True)

    if not ranking:
        box.innerHTML = '<div class="empty">아직 기록이 없습니다.</div>'
        return

    box.innerHTML = ""

    for i,p in enumerate(ranking):
        n = i+1
        row = document.createElement("div")
        row.className = "player " + {1:"first",2:"second",3:"third"}.get(n,"")

        rank = "🥇" if n==1 else "🥈" if n==2 else "🥉" if n==3 else str(n)

        row.innerHTML = f'''
            <div class="rank">{rank}</div>
            <div class="name"></div>
            <div class="score">{p["score"]}점</div>
            <button class="del">삭제</button>
        '''

        row.querySelector(".name").innerText = p["name"]
        row.querySelector(".del").onclick = lambda e,i=i: delete(i)
        box.appendChild(row)

def add(event):
    name = document.getElementById("name")
    score = document.getElementById("score")
    msg = document.getElementById("msg")

    if not name.value.strip():
        msg.innerText = "닉네임을 입력하세요."
        return

    if not score.value.strip():
        msg.innerText = "점수를 입력하세요."
        return

    try:
        s = int(score.value)
    except:
        msg.innerText = "점수는 숫자로 입력하세요."
        return

    ranking.append({"name":name.value.strip(),"score":s})
    save()
    render()

    name.value = ""
    score.value = ""
    msg.innerText = ""

def delete(i):
    ranking.pop(i)
    save()
    render()

def clear(event):
    global ranking
    ranking = []
    save()
    render()

render()
